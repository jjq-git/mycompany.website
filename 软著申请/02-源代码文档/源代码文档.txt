def _ensure_page_detail_templates(site_id, dataset_key, label=None, lang=None):

        page['detail_templates'] = [_default_detail_template(dataset_key, label)]
        changed = True
    else:
        has_default = any(

            for tpl in templates

def _parse_tags_input(value):
    """Parse the admin tags input without breaking phrases that contain spaces."""
    if value is None:
        return []
    if isinstance(value, (list, tuple)):
        raw_parts = value
    else:
        raw = str(value).strip()
        if not raw:
            return []
        raw_parts = re.split(r'[,，;；\n\r]+', raw)
    result = []
    seen = set()
    for part in raw_parts:
        tag = str(part or '').strip()
        if not tag or tag in seen:
            continue
        seen.add(tag)
        result.append(tag)
    return result

def _hex_to_rgb(hex_color):
    """将 #rrggbb 转为 (r, g, b) 整数元组；无效输入返回黑色 (0, 0, 0)."""
    h = hex_color.lstrip('#')
    if len(h) < 6:
        return (0, 0, 0)
    try:
        return tuple(int(h[i:i+2], 16) for i in (0, 2, 4))
    except (ValueError, IndexError):
        return (0, 0, 0)


def _darken(hex_color, factor=0.1):
    """将 hex 颜色变暗 factor (0-1)."""
    r, g, b = _hex_to_rgb(hex_color)
    return '#{:02x}{:02x}{:02x}'.format(
        max(0, int(r * (1 - factor))),
        max(0, int(g * (1 - factor))),
        max(0, int(b * (1 - factor))),
    )


def _lighten(hex_color, factor=0.1):
    """将 hex 颜色变亮 factor (0-1)."""
    r, g, b = _hex_to_rgb(hex_color)
    return '#{:02x}{:02x}{:02x}'.format(
        min(255, int(r + (255 - r) * factor)),
        min(255, int(g + (255 - g) * factor)),
        min(255, int(b + (255 - b) * factor)),
    )

def _safe_send_from_directory(base_dir, relative_path):
    """加固版 send_from_directory: 解析 symlink 后强制校验目标仍在 base_dir 内,
    阻断符号链接逃逸 + 兜底防御 (Flask safe_join 自身也已经防 ../, 这里多一层保险).
    """
    base_real = Path(base_dir).resolve()
    full_real = (base_real / relative_path).resolve()
    try:
        full_real.relative_to(base_real)
    except ValueError:
        abort(403, '非法路径')
    return send_from_directory(base_real, relative_path)

def _consume_verification_code(identifier, purpose, code):
    record = VerificationCode.query.filter_by(
        identifier=identifier, purpose=purpose, consumed_at=None
    ).order_by(VerificationCode.id.desc()).first()
    if not record or record.attempt_count >= 5:
        return None
    record.attempt_count += 1
    now = datetime.now(timezone.utc)
    expires = record.expires_at
    if expires.tzinfo is None:
        expires = expires.replace(tzinfo=timezone.utc)
    if expires < now or not secrets.compare_digest(record.code_hash, _hash_token((code or '').strip())):
        db.session.commit()
        return None
    record.consumed_at = now
    db.session.commit()
    return record

    """通过 stream.seek 获取真实字节数, 超限直接报错."""
    stream = file_storage.stream
    try:
        stream.seek(0, 2)
        size = stream.tell()
    finally:
        stream.seek(0)
    if size > max_bytes:
        raise ValueError(f'{label}过大: {size // 1024}KB > {max_bytes // 1024}KB 上限')
    return size

def _context_alt(*contexts, fallback=''):
    for ctx in contexts:
        if not isinstance(ctx, dict):
            continue
        for key in ('title', 'name', 'label', 'short_name', 'heading'):
            val = ctx.get(key)
            if isinstance(val, str) and val.strip():
                return val.strip()
    return str(fallback or '').strip()

def _safe_zip_member_name(name):
    name = (name or '').replace('\\', '/').strip('/')
    if not name or name.startswith('.') or '\x00' in name:
        raise ValueError('网页包中包含非法文件名')
    path = Path(name)
    if path.is_absolute() or any(part in ('', '.', '..') or part.startswith('.') for part in path.parts):
        raise ValueError('网页包中包含非法路径')
    if len(path.parts) > MAX_MEDIA_PACKAGE_DEPTH:
        raise ValueError('网页包目录层级过深')
    return name


def _zip_strip_root(member_names):
    first_parts = [Path(name).parts for name in member_names]
    if not first_parts:
        return ''
    first = first_parts[0][0]
    if all(len(parts) > 1 and parts[0] == first for parts in first_parts):
        return first
    return ''

def _current_user():
    """从 session 读取当前用户；无效 session 自动清理."""
    user_id = session.get('user_id')
    if not user_id:
        return None
    user = User.query.get(user_id)
    if not user or user.status != 'active':
        session.pop('user_id', None)
        return None
    raw_token = session.get('auth_session_token')
    if raw_token:
        login_session = UserSession.query.filter_by(token_hash=_hash_token(raw_token), user_id=user.id).first()
        if not login_session or login_session.revoked_at:
            session.clear()
            return None
        expires = login_session.expires_at
        if expires.tzinfo is None:
            expires = expires.replace(tzinfo=timezone.utc)
        if expires < datetime.now(timezone.utc):
            session.clear()
            return None
        # 限制写频率，避免每个静态资源请求都更新数据库。
        last_seen = login_session.last_seen_at
        if last_seen.tzinfo is None:
            last_seen = last_seen.replace(tzinfo=timezone.utc)
        if datetime.now(timezone.utc) - last_seen > timedelta(minutes=5):
            login_session.last_seen_at = datetime.now(timezone.utc)
            db.session.commit()
    return user

def _repair_truncated_json(raw):
    """尝试修复因 max_tokens 截断而残缺的 JSON。

    常见截断模式（从易到难）：
    1. 缺少末尾的 "} → 补上
    2. 最后一行的字符串值未闭合 → 去掉最后一行
    3. 中间某行的字符串值未闭合 → 补闭合引号

    返回修复后的 dict，无法修复返回 None。
    """
    if not raw or not raw.strip():
        return None

    text = raw.strip()

    # 策略 1：逐行扫描，修复未闭合的字符串
    lines = text.split('\n')
    fixed_lines = []
    for i, line in enumerate(lines):
        stripped = line.rstrip()
        # 统计引号：值为 JSON 字符串时引号成对，不成对说明被截断
        in_str = False
        escape_next = False
        for ch in stripped:
            if escape_next:
                escape_next = False
                continue
            if ch == '\\':
                escape_next = True
                continue
            if ch == '"':
                in_str = not in_str
        # 如果这行在字符串中结束（引号不对）→ 这是截断行
        if in_str:
            # 行内字符串被截断 → 丢弃此行及之后内容，只保留前面的行
            text = '\n'.join(fixed_lines).rstrip()
            break
        fixed_lines.append(stripped)

    # 策略 2：补全缺失的括号
    # 去掉末尾逗号后再补括号

======================================================================
                         小企业静态网站生成与管理平台  V1.0                         
                                                       第 33 页 / 共 60 页
======================================================================
    text = text.rstrip().rstrip(',').rstrip()

    # 数括号
    open_braces = text.count('{') - text.count('}')
    open_brackets = text.count('[') - text.count(']')

    # 补缺失的闭合括号
    text += ']' * open_brackets
    text += '}' * open_braces

    try:
        result = json.loads(text)
        if isinstance(result, dict):
            return result
    except json.JSONDecodeError:
        pass

    # 策略 3：去掉最后一行再试（常见：最后一行不完整）
    if len(lines) > 2:
        text = '\n'.join(lines[:-1]).rstrip().rstrip(',').rstrip()
        open_braces = text.count('{') - text.count('}')
        open_brackets = text.count('[') - text.count(']')
        text += ']' * open_brackets
        text += '}' * open_braces
        try:
            result = json.loads(text)
            if isinstance(result, dict):
                return result
        except json.JSONDecodeError:
            pass

    return None

def _split_batches(texts):
    """将 texts 字典按 _BATCH_MAX_CHARS 拆成多个小批，每批仍是 dict。

    拆批策略：贪心累加，当前批 JSON 序列化后不超过阈值。单个 key-value 超阈值
    也不拆分（一个 value 必须完整给 API），此时该批会暂时超出阈值。
    """
    batches = []
    current = {}
    current_size = 0
    # 估算 JSON 开销：大括号、key 的引号冒号、逗号等，每 key 约 10 字符
    overhead_per_key = 10

    for key, value in texts.items():
        value_str = str(value) if value else ''
        # 预估该条目在 JSON 中的大小：key 长度 + value 长度 + 引号等开销
        item_size = len(key) + len(value_str) + overhead_per_key

        if current and current_size + item_size > _BATCH_MAX_CHARS:
            batches.append(current)
            current = {}
            current_size = 0

        current[key] = value
        current_size += item_size

    if current:
        batches.append(current)

    return batches

def _translate_one_batch(api_key, api_base, model, batch, source_lang, target_lang):
    """翻译单批，JSON 截断时自动拆半重试（最多 2 层）。"""

    except RuntimeError as e:
        msg = str(e)
        # 只对 JSON 截断/格式异常做拆分重试，网络/认证错误不重试
        if '格式异常' not in msg:
            raise

    # 批次有 2 个以上字段且 JSON 被截断 → 拆成两半分别翻译
    keys = list(batch.keys())
    if len(keys) < 2:
        raise  # 单字段还失败，没法拆了

    mid = len(keys) // 2
    left = {k: batch[k] for k in keys[:mid]}
    right = {k: batch[k] for k in keys[mid:]}

    result = {}
    for label, sub in [('左半', left), ('右半', right)]:
        try:

        return ''.join(n.data for n in self.nodes if isinstance(n, _Text)).strip()

class _DomBuilder(HTMLParser):
    """把 HTML 流还原成 _Node 树。维护一个元素栈处理父子关系。

    convert_charrefs=False：保留属性/文本里原始的 HTML 实体（&amp; &quot; 等），
    序列化时不会被解码后又裸写出去破坏 HTML（见 _serialize 的转义）。
    """

    def __init__(self):
        super().__init__(convert_charrefs=False)
        self.root = None
        self._stack = []

    def _append(self, child):
        if self._stack:
            self._stack[-1].nodes.append(child)
        elif isinstance(child, _Node) and self.root is None:
            self.root = child

    def handle_starttag(self, tag, attrs):
        node = _Node(tag=tag, attrs=dict(attrs))
        self._append(node)
        # void 元素没有结束标签，不入栈
        if tag.lower() not in VOID_TAGS:
            self._stack.append(node)

    def handle_startendtag(self, tag, attrs):
        # 显式自闭合 <x/>：建节点但不入栈
        self._append(_Node(tag=tag, attrs=dict(attrs), self_closing=True))

    def handle_endtag(self, tag):
        # 容错：找到栈中最近的同名标签弹出（应对未闭合的中间标签）
        for i in range(len(self._stack) - 1, -1, -1):
            if self._stack[i].tag.lower() == tag.lower():

======================================================================
                         小企业静态网站生成与管理平台  V1.0                         
                                                       第 42 页 / 共 60 页
======================================================================
                del self._stack[i:]
                break

    def handle_data(self, data):
        # script/style 内是 CDATA：HTMLParser 给的是原始未解析文本（含 < > &），
        # 必须原样输出不转义，否则会破坏 JS（如 a < b 变 a &lt; b）。
        in_cdata = bool(self._stack) and self._stack[-1].tag.lower() in ('script', 'style')
        self._append(_Text(data, raw=in_cdata))

    def handle_entityref(self, name):
        # convert_charrefs=False 下命名实体走这里：原样保留 &name;
        self._append(_Text(f'&{name};', raw=True))

    def handle_charref(self, name):
        # 数字字符引用：原样保留 &#...;
        self._append(_Text(f'&#{name};', raw=True))


def _serialize_attrs(attrs, in_svg=False):
    """序列化属性。值统一转义（quote=True），无值属性输出裸名（如 hidden）。

    当 in_svg=True 时，将 HTMLParser 转小写后的 SVG camelCase 属性名恢复正确大小写
    （如 viewbox→viewBox），防止内联 SVG 图标因属性名错误无法渲染。
    """
    out = ''
    for k, v in attrs.items():
        if in_svg and k in _SVG_ATTR_CASE:
            k = _SVG_ATTR_CASE[k]
        if v is None:
            out += f' {k}'
        else:
            out += f' {k}="{escape(str(v), quote=True)}"'
    return out


def _serialize(node, in_svg=False):
    """把 _Node 树重新序列化为 HTML 字符串。

    - 子节点按 nodes 的原始顺序输出（文本与元素交错不重排）
    - 属性值转义，无值属性保留（hidden）
    - 文本节点：raw（实体/富文本）原样输出，普通文本转义裸 < > &
    - in_svg 追踪 SVG 上下文：进入 <svg> 标签后其子孙的属性名恢复 camelCase
    """
    tag = node.tag.lower()
    is_svg = in_svg or tag in _SVG_TAGS

    attrs = _serialize_attrs(node.attrs, is_svg or tag == 'svg')
    if tag in VOID_TAGS or node.self_closing:
        return f'<{node.tag}{attrs}>'
    inner = ''

======================================================================
                         小企业静态网站生成与管理平台  V1.0                         
                                                       第 43 页 / 共 60 页
======================================================================
    for child in node.nodes:
        if isinstance(child, _Text):
            inner += child.data if child.raw else escape(child.data, quote=False)
        else:
            inner += _serialize(child, is_svg)
    return f'<{node.tag}{attrs}>{inner}</{node.tag}>'


def _direct_named_children(node):
    """返回 node 的"带 data-name 的直接子元素"。

    规范：list/grid 容器内带 data-name 的直接子元素 = 项模板。
    不带 data-name 的杂项（标题、"加载更多"按钮）会被跳过、不递归进去找。
    """
    result = []
    for child in node.children:
        if 'data-name' in child.attrs:
            result.append(child)
        else:
            # 中间包裹层（无 data-name）：向下找一层，应对 <ul><li data-name>结构
            result.extend(_direct_named_children(child))
    return result


def _int_or_none(raw):
    try:
        return int(raw)
    except (TypeError, ValueError):
        return None


def _style_tokens(attrs):
    """解析一个元素上的 data-style，只保留注册表里已声明的 token。"""
    return [
        t for t in style_registry.parse_style_tokens(attrs.get('data-style', ''))
        if style_registry.get_style_def(t)
    ]


def _descendant_style_tokens(node):
    """汇总子孙元素上的 data-style token，供编辑器生成样式控件。

    普通包裹层可以只写 data-style，不必强行变成 data-name/data-type 字段。
    """
    tokens = []
    seen = set()

    def walk(current):
        for child in current.children:
            for token in _style_tokens(child.attrs):

                         小企业静态网站生成与管理平台  V1.0

======================================================================
                if token not in seen:
                    seen.add(token)
                    tokens.append(token)
            walk(child)

    walk(node)
    return tokens


def _item_children(item):
    """解析 list/grid 项模板的可编辑子字段（供编辑器生成条目子表单）。

    两种项形态：
    - 项根是 group（如 feature 卡片）：取其带 data-name 的直接子元素为子字段
    - 项根本身就是 leaf（如 logo 墙的 <img data-name="logo">）：项根自己即唯一子字段，
      渲染器走 _inject_root_leaf 不读 children，这里把它暴露出来纯粹是给编辑器用
    """
    if item.attrs.get('data-type', '') in LEAF_TYPES:
        return [_build_field(item)]
    return [_build_field(child) for child in _direct_named_children(item)]


def _build_field(node):
    """把一个带 data-name 的 _Node 转成 FieldDef，递归处理子字段。"""
    attrs = node.attrs
    dtype = attrs.get('data-type', '')
    fd = FieldDef(
        name=attrs.get('data-name', ''),
        type=dtype,
        tag=node.tag,
        href_field=attrs.get('data-href', '') or '',
        href_template=attrs.get('data-href-template', '') or '',
        fallback=[k.strip() for k in (attrs.get('data-fallback') or '').split(',') if k.strip()],
        hideable='data-hideable' in attrs,
        style_tokens=_style_tokens(attrs),
    )

    if dtype in LEAF_TYPES:
        # 叶子：文本类型存 innerText，媒体类型存 src
        if dtype in ('text', 'textarea', 'richtext', 'icon'):
            fd.default_text = node.text
        if dtype in ('image', 'video', 'audio', 'embed'):
            fd.default_src = attrs.get('src', '')
        return fd

    if dtype == 'group':
        # 复合项：递归解析子字段
        for child in _direct_named_children(node):
            fd.children.append(_build_field(child))
        return fd

                                                       第 45 页 / 共 60 页
======================================================================

    if dtype in LIST_TYPES:
        # 可重复列表/栅格
        fd.source = attrs.get('data-source', None)
        # data-source 存在但写成属性无值时，HTMLParser 给 None；规范里"存在即数据驱动"
        if 'data-source' in attrs and fd.source is None:
            fd.source = ''
        fd.page_size = _int_or_none(attrs.get('data-page-size'))
        fd.cols = _int_or_none(attrs.get('data-cols'))
        # 项模板 = 第一个带 data-name 的直接子元素
        items = _direct_named_children(node)
        if items:
            item = items[0]
            fd.item_template_html = _serialize(item)
            # 项模板内部的字段也解析出来（编辑器要用）
            fd.children.extend(_item_children(item))
        return fd

    # 未知/未声明 type 的带名元素：当文本叶子兜底
    fd.default_text = node.text
    return fd


def parse(html):
    """解析 block 模板 HTML，返回 BlockSchema。

    html: block 的 template.html 内容（纯 HTML，不含 Jinja2 标记）。
    """
    builder = _DomBuilder()
    builder.feed(html)
    builder.close()
    root = builder.root
    if root is None:
        raise ValueError('block 模板为空或无法解析出根元素')

    attrs = root.attrs
    schema = BlockSchema(
        block_name=attrs.get('data-name', ''),
        root_tag=root.tag,
        root_type=attrs.get('data-type', ''),
        hideable='data-hideable' in attrs,
        href_field=attrs.get('data-href', '') or '',
        style_tokens=_style_tokens(attrs),
        element_style_tokens=_descendant_style_tokens(root),
        from_source=attrs.get('data-from', None) or None,
    )

    root_type = schema.root_type
    if root_type in LIST_TYPES:
        schema.source = attrs.get('data-source', None)

======================================================================
                         小企业静态网站生成与管理平台  V1.0                         
                                                       第 46 页 / 共 60 页
======================================================================
        if 'data-source' in attrs and schema.source is None:
            schema.source = ''
        schema.page_size = _int_or_none(attrs.get('data-page-size'))
        schema.cols = _int_or_none(attrs.get('data-cols'))
        items = _direct_named_children(root)
        if items:
            item = items[0]
            schema.item_template_html = _serialize(item)
            schema.fields.extend(_item_children(item))
    else:
        # group 或其他：直接子字段
        for child in _direct_named_children(root):
            schema.fields.append(_build_field(child))

    return schema


def is_data_attr_block(html):
    """判断一段 block 模板是否为 data-* 声明式写法.

    判据（对齐方案 4.2）：含 Jinja2 标记（{% 或 {{）= 旧式命令式；
    否则视为纯 HTML 声明式。空内容当旧式处理（保守）。
    """
    if not html:
        return False
    if '{%' in html or '{{' in html:
        return False
    return 'data-name' in html

def check_upload_size(file_storage, max_bytes, label='文件'):
    stream = file_storage.stream
    try:
        stream.seek(0, 2)
        size = stream.tell()
    finally:
        stream.seek(0)

        raise ValueError(f'{label}过大: {size // 1024}KB > {max_bytes // 1024}KB 上限')

    field_name = getattr(file_storage, 'name', '') or ''

    if field_name:

def image_alt_key(field_name):
    field = str(field_name or '').strip()
    if not field:
        return 'alt'
    if field in ('cover', 'photo', 'logo'):
        return f'{field}_alt'
    if field in ('image', 'media', 'url', 'src'):
        return 'alt'
    if field.endswith('_url'):
        return f'{field[:-4]}_alt'
    return f'{field}_alt'


def form_alt_value(form, field_name, *contexts, upload=None, existing=None):
    keys = []
    field = str(field_name or '').strip()

        keys.extend([f'{field}_alt', image_alt_key(field)])
    keys.append('alt')
    for key in dict.fromkeys(keys):
        if key in form:
            val = form.get(key, '').strip()
            if val:
                return val
    if existing:
        return str(existing).strip()
    return context_alt(*contexts, fallback=filename_alt(upload))

def scan_templates():
    """扫描 site_templates/, 把每个含 meta.yaml 的子目录登记到数据库.

    返回: 同步后所有 Template 实例的列表.
    """
    templates_dir = Path(current_app.config['TEMPLATES_DIR'])
    if not templates_dir.exists():
        return []

    found = []
    for folder in sorted(templates_dir.iterdir()):
        if not folder.is_dir() or folder.name.startswith('.'):
            continue

        meta_path = folder / 'meta.yaml'
        if not meta_path.exists():
            continue

        meta = yaml.safe_load(meta_path.read_text(encoding='utf-8')) or {}

        template = Template.query.filter_by(name=folder.name).first()
        if template is None:
            template = Template(name=folder.name)

def scan_blocks(template_name):
    """扫描模板的 blocks/ 目录, 返回模块列表.

    返回: list of dict, 每个模块:
        name, category, icon, description, fields, variants, is_data_attr, schema

    is_data_attr / schema: 若 template.html 为 data-* 声明式写法（不含 Jinja2 标记），
    附加解析出的 schema（供编辑器 schema 驱动表单）；旧 Jinja2 block 这两项为 False/None。
    """
    blocks_dir = Path(current_app.config['TEMPLATES_DIR']) / template_name / 'blocks'
    if not blocks_dir.exists():
        return []

    blocks = []
    for block_dir in sorted(blocks_dir.iterdir()):
        if not block_dir.is_dir() or block_dir.name.startswith('.'):
            continue
        meta_path = block_dir / 'meta.yaml'
        if not meta_path.exists():
            continue

        meta = yaml.safe_load(meta_path.read_text(encoding='utf-8')) or {}

        # 检测 template.html 是否为 data-* 声明式，是则解析 schema。
        # 解析失败（如旧式 Jinja2）静默降级，不影响扫描——混合模式共存。
        is_data_attr = False
        schema = None

======================================================================
                         小企业静态网站生成与管理平台  V1.0                         
                                                       第 55 页 / 共 60 页
======================================================================
        tpl_path = block_dir / 'template.html'
        if tpl_path.exists():
            html = tpl_path.read_text(encoding='utf-8')
            if block_parser.is_data_attr_block(html):
                try:
                    schema = block_parser.parse(html)
                    is_data_attr = True
                except Exception:
                    is_data_attr = False
                    schema = None

        # 自动从 data-style 派生样式变体（select/switch 类 control）。
        # 主 template.html 和 variants/*.html 都可能声明 data-style；meta.yaml 声明
        # 的结构变体优先，可覆盖同 key 的样式变体。
        meta_variants = meta.get('variants', [])
        meta_variant_keys = {v['key'] for v in meta_variants if isinstance(v, dict)}
        style_variants = []

        def _append_tokens_from_schema(parsed_schema, tokens):
            if parsed_schema is None:
                return
            for t in list(parsed_schema.style_tokens or []):
                if t not in tokens:
                    tokens.append(t)
            for t in getattr(parsed_schema, 'element_style_tokens', []) or []:
                if t not in tokens:
                    tokens.append(t)

        def _style_variants_for_tokens(tokens, base_variant=None, base_label=''):
            variants = []
            for token in tokens:
                sdef = style_registry.get_style_def(token)
                if not sdef or sdef.get('control') not in ('select', 'switch'):
                    continue
                for opt in sdef.get('options', []):
                    opt_value = opt['value']
                    key = f'{token}_{opt_value}'
                    label = opt.get('text', opt_value)
                    if base_variant:
                        key = f'{base_variant}_{key}'
                        label = f'{base_label or base_variant} / {label}'
                    variant = {
                        'key': key,
                        'label': label,
                        'source': 'style',
                        'data_preset': {f'style_{token}': opt_value},
                    }
                    if base_variant:
                        variant['base_variant'] = base_variant
                    variants.append(variant)

                         小企业静态网站生成与管理平台  V1.0                         
                                                       第 56 页 / 共 60 页
======================================================================
            return variants

        all_tokens = []
        _append_tokens_from_schema(schema, all_tokens)
        style_variants.extend(_style_variants_for_tokens(all_tokens))

        variants_dir = block_dir / 'variants'
        if variants_dir.exists():
            meta_variant_labels = {
                v.get('key'): v.get('label', v.get('key'))
                for v in meta_variants
                if isinstance(v, dict) and v.get('key')
            }
            for variant_tpl in sorted(variants_dir.glob('*.html')):
                try:
                    variant_html = variant_tpl.read_text(encoding='utf-8')
                    if block_parser.is_data_attr_block(variant_html):
                        variant_tokens = []
                        _append_tokens_from_schema(block_parser.parse(variant_html), variant_tokens)
                        base_variant = variant_tpl.stem
                        style_variants.extend(_style_variants_for_tokens(
                            variant_tokens,
                            base_variant=base_variant,
                            base_label=meta_variant_labels.get(base_variant, base_variant),
                        ))
                except Exception:
                    current_app.logger.exception(
                        '解析 block 变体样式失败: %s/%s/%s',
                        template_name, block_dir.name, variant_tpl.name
                    )
        all_variants = meta_variants + [
            v for v in style_variants if v['key'] not in meta_variant_keys
        ]

        blocks.append({
            'name': block_dir.name,
            'display_name': meta.get('name', block_dir.name),
            'category': meta.get('category', 'content'),
            'icon': meta.get('icon', ''),
            'description': meta.get('description', ''),
            'fields': meta.get('fields', []),
            'variants': all_variants,
            'is_data_attr': is_data_attr,
            'schema': asdict(schema) if schema is not None else None,
        })

    return blocks

    """只解析单个 block 的 schema（不扫描整个目录）。

        return None

"""CSS 变量全局注册表.

设计要点（对齐 docs/docs/data-attributes-reference.md 第五节）：
- 按"变量"建表，不是按"block"建表。features 和 news 都用 cols → 共用一条。
- 默认值只写在模板 CSS 的 var(--xxx, 默认) 里，这里和 HTML 都不重复声明默认值。
  （理由：默认值多处声明必然不同步；CSS 是唯一真相源。）
- token 名 ↔ CSS 变量名一一对应，加 -- 前缀。data-style="dir" ↔ --dir。

后端用途：渲染器读元素的 data-style → 查本表确认是合法 token → 注入用户选的值

"""
